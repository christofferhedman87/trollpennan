import fs from "fs";

const base = m.tolka("en kanin");
const baseSig = JSON.stringify({...base.varelse, formel:0, beskrivning:0});
const sig = u => u.typ==="skapad" ? JSON.stringify({...u.varelse, formel:0, beskrivning:0}) : null;
function pres(inf){ // heuristik: -a -> -ar/-er/-r
  const s = inf.endsWith("a") ? inf.slice(0,-1) : inf;
  return [...new Set([s+"ar", s+"er", s+"r", inf+"r", inf])];
}
function testNoun(w){
  const r = {};
  const a = m.tolka(w);
  if (a.typ==="skapad" && !a.okanda.includes(w)) return {k:"djur/sammansatt", via:"tolka"};
  const b = m.tolka("en kanin med "+w);
  if (b.typ==="skapad" && !b.okanda.includes(w) && sig(b)!==baseSig) return {k:"tillbehor/monster", via:"med"};
  const c = m.tolkaBygg("ett "+w);
  if (c.typ==="bygge" && !c.okanda.includes(w)) return {k:"bygg", via:"bygg"};
  if (a.typ==="forslag"||b.typ==="forslag"||c.typ==="forslag") return {k:"forslag"};
  return {k:"okand"};
}
function testAdj(w){
  const u = m.tolka("en "+w+" kanin");
  if (u.typ==="skapad" && !u.okanda.includes(w) && sig(u)!==baseSig) return {k:"synlig"};
  if (u.typ==="forslag") return {k:"forslag"};
  return {k:"okand"};
}
function testVerb(w){
  const forms = [["en kanin som kan "+w]];
  for (const p of pres(w)) forms.push(["en kanin som "+p]);
  let forslag=false;
  for (const [f] of forms){
    const u = m.tolka(f);
    if (u.typ==="skapad" && u.okanda.length===0 && sig(u)!==baseSig) return {k:"synlig"};
    if (u.typ==="forslag") forslag=true;
  }
  return forslag?{k:"forslag"}:{k:"okand"};
}
const out = {};
function run(name, list, fn){
  const res = list.map(r=>({w:clean(r.gf), wpm:r.wpm, cefr:r.cefr, ...fn(clean(r.gf))}));
  const cnt = {}; let wsum=0, tot=0;
  for (const x of res){ cnt[x.k]=(cnt[x.k]||0)+1; tot+=x.wpm; if(!["okand","forslag"].includes(x.k)) wsum+=x.wpm; }
  out[name]={n:list.length,cnt,covered:res.filter(x=>!["okand","forslag"].includes(x.k)).length, wpmShare:+(wsum/tot*100).toFixed(1), träffar:res.filter(x=>!["okand","forslag"].includes(x.k)).map(x=>x.w+":"+x.k), forslag:res.filter(x=>x.k==="forslag").map(x=>x.w)};
  return res;
}
const top = (p,n)=>sw.filter(pos[p]).sort((a,b)=>b.wpm-a.wpm).slice(0,n);
const fn = {noun:testNoun, adj:testAdj, verb:testVerb};
const all = {};
for (const p of ["noun","adj","verb"]) {
  all["top1000-"+p]=run("top1000-"+p, top(p,1000), fn[p]);
  const a12 = rows.filter(pos[p]).filter(r=>["1","2"].includes(r.cefr));
  all["A1A2-"+p]=run("A1A2-"+p, a12, fn[p]);
  all["top100-"+p]=run("top100-"+p, top(p,100), fn[p]);
}
fs.writeFileSync("resultat.json", JSON.stringify({out, all},null,1));
for (const [k,v] of Object.entries(out)) console.log(k, v.n, JSON.stringify(v.cnt), "täckt", v.covered, (v.covered/v.n*100).toFixed(1)+"%", "ordfrekvensandel", v.wpmShare+"%", "\n   träffar:", v.träffar.slice(0,40).join(" "), "\n   forslag:", v.forslag.slice(0,12).join(" "));
